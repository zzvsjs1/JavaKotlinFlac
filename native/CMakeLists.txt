cmake_minimum_required(VERSION 3.20)
project(jflac_jni C)

find_package(JNI REQUIRED)

set(JFLAC_FLAC_INCLUDE_DIR "" CACHE PATH "Path to the FLAC public include directory")
if(NOT JFLAC_FLAC_INCLUDE_DIR)
    set(JFLAC_FLAC_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../flac-1.5.0/include")
endif()

set(JFLAC_GENERATED_JNI_INCLUDE_DIR "" CACHE PATH "Path to the javac-generated JNI header directory")
if(NOT JFLAC_GENERATED_JNI_INCLUDE_DIR)
    message(FATAL_ERROR "JFLAC_GENERATED_JNI_INCLUDE_DIR must point to the JNI headers generated by javac.")
endif()
if(NOT EXISTS "${JFLAC_GENERATED_JNI_INCLUDE_DIR}/org_zzvsjs_jflac_internal_NativeBindings.h")
    message(FATAL_ERROR "The javac-generated NativeBindings JNI header is missing.")
endif()

set(JFLAC_EXPECTED_FLAC_VERSION "" CACHE STRING "FLAC runtime version expected by the JNI shim")
if(NOT JFLAC_EXPECTED_FLAC_VERSION)
    message(FATAL_ERROR "JFLAC_EXPECTED_FLAC_VERSION must name the FLAC runtime version bundled with the JNI shim.")
endif()

set(JFLAC_EXPECTED_FLAC_API_VERSION_CURRENT "" CACHE STRING "Expected FLAC API current version")
set(JFLAC_EXPECTED_FLAC_API_VERSION_REVISION "" CACHE STRING "Expected FLAC API revision")
set(JFLAC_EXPECTED_FLAC_API_VERSION_AGE "" CACHE STRING "Expected FLAC API age")
if(JFLAC_EXPECTED_FLAC_API_VERSION_CURRENT STREQUAL "" OR
   JFLAC_EXPECTED_FLAC_API_VERSION_REVISION STREQUAL "" OR
   JFLAC_EXPECTED_FLAC_API_VERSION_AGE STREQUAL "")
    message(FATAL_ERROR "All expected FLAC API version components must be provided.")
endif()

if(WIN32)
    set(JFLAC_PLATFORM_SOURCE src/jflac_platform_windows.c)
    set(JFLAC_FLAC_LIBRARY_FILENAME "FLAC.dll")
elseif(APPLE)
    set(JFLAC_PLATFORM_SOURCE src/jflac_platform_posix.c)
    set(JFLAC_FLAC_LIBRARY_FILENAME "libFLAC.14.dylib")
else()
    set(JFLAC_PLATFORM_SOURCE src/jflac_platform_posix.c)
    set(JFLAC_FLAC_LIBRARY_FILENAME "libFLAC.so.14")
endif()

add_library(jflac-jni SHARED
    src/jflac_jni.c
    ${JFLAC_PLATFORM_SOURCE}
)

set_target_properties(jflac-jni PROPERTIES
    C_STANDARD 11
    C_STANDARD_REQUIRED YES
    C_EXTENSIONS NO
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
)

target_include_directories(jflac-jni PRIVATE
    ${JNI_INCLUDE_DIRS}
    "${JFLAC_GENERATED_JNI_INCLUDE_DIR}"
    "${JFLAC_FLAC_INCLUDE_DIR}"
)

target_compile_definitions(jflac-jni PRIVATE
    JFLAC_EXPECTED_FLAC_VERSION="${JFLAC_EXPECTED_FLAC_VERSION}"
    JFLAC_EXPECTED_FLAC_API_VERSION_CURRENT=${JFLAC_EXPECTED_FLAC_API_VERSION_CURRENT}
    JFLAC_EXPECTED_FLAC_API_VERSION_REVISION=${JFLAC_EXPECTED_FLAC_API_VERSION_REVISION}
    JFLAC_EXPECTED_FLAC_API_VERSION_AGE=${JFLAC_EXPECTED_FLAC_API_VERSION_AGE}
    JFLAC_FLAC_LIBRARY_FILENAME="${JFLAC_FLAC_LIBRARY_FILENAME}"
)

if(MSVC)
    target_compile_definitions(jflac-jni PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(jflac-jni PRIVATE /W4 /utf-8)
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(jflac-jni PRIVATE -Wall -Wextra)
endif()

if(WIN32)
    target_link_libraries(jflac-jni PRIVATE kernel32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(jflac-jni PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
